Q1 
Display customer full name in UPPERCASE and show their credit limit. 
If credit_limit is NULL, display 0 instead. 
Sort by credit_limit descending. 

SELECT UPPER(CONCAT(first_name,' ',last_name)) AS full_name,
IFNULL(credit_limit,0) AS credit_limit
FROM customers
ORDER BY credit_limit DESC;

Q2 
List all orders placed in 2023 where: 
 order_status is 'SHIPPED' 
 AND shipped_date is more than 3 days after order_date. 

SELECT * 
FROM orders
WHERE YEAR(order_date) = 2023 
	AND order_status = 'SHIPPED'
	AND DATEDIFF(shipped_date, order_date) > 3;

Q8 
Display product category and total quantity sold per category. 
Show only categories where total quantity sold exceeds 50. 
Sort by total quantity descending.

SUM(o.quantity) AS total_qty
FROM order_items o
JOIN products p ON o.product_id=p.product_id

HAVING SUM(o.quantity) > 50

Q11

 'LOW' if stock_qty < 20

					AND last_name='Lewis'
					)

Q24 
Display customers who have ordered products from category 'Electronics' 
but never ordered products from category 'Furniture'.

SELECT DISTINCT c.customer_id
FROM customers c
JOIN orders o ON c.customer_id=o.customer_id
JOIN order_items oi ON o.order_id=oi.order_id
JOIN products p ON oi.product_id=p.product_id
WHERE p.category='Electronics'
AND c.customer_id NOT IN
			(
			SELECT DISTINCT c.customer_id
			FROM customers c
			JOIN orders o ON c.customer_id=o.customer_id
			JOIN order_items oi ON o.order_id=oi.order_id
			JOIN products p ON oi.product_id=p.product_id
			WHERE p.category='Furniture'
			);

Display product categories where the total revenue generated is 
greater than the overall average revenue per category. 

SELECT category, SUM(quantity*unit_price-discount) AS revenue
FROM order_items oi
JOIN products p ON oi.product_id=p.product_id
GROUP BY category
HAVING revenue >
		(
		SELECT AVG(rev)
		FROM
			(
			SELECT SUM(quantity*unit_price-discount) AS rev
			FROM order_items oi
			JOIN products p ON oi.product_id=p.product_id
			GROUP BY category
			) t
		);
